# Kubernetes Commands

kubectl version

kubectl get nodes

kubectl get pods -A

kubectl cluster-info

# Kubernetes Resource Requests and Limits

# Create Pod

kubectl apply -f requests-limits.yaml

# View Pods

kubectl get pods

# Describe Pod

kubectl describe pod nginx-resources

# Delete Pod

kubectl delete pod nginx-resources

# Validate YAML

kubectl apply --dry-run=client -f requests-limits.yaml

# View Pod Resource Configuration

kubectl describe pod nginx-resources

# View Events

kubectl get events

# View Node Capacity

kubectl describe node <node-name>

# View Resource Usage (Metrics Server Required)

kubectl top pod
kubectl top node


# Kubernetes Probes

# Create Pod With Probes

kubectl apply -f probes.yaml

# View Pods

kubectl get pods

# Describe Pod

kubectl describe pod nginx-probes

# Delete Pod

kubectl delete pod nginx-probes

# View Events

kubectl get events

# Validate YAML

kubectl apply --dry-run=client -f probes.yaml

# Check Pod Status

kubectl get pod nginx-probes

# Watch Pod Continuously

kubectl get pod nginx-probes -w

# View Logs

kubectl logs nginx-probes

# View Detailed Pod Information

kubectl describe pod nginx-probes
